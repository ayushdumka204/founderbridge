);

-- Prevent users from self-assigning admin role
CREATE OR REPLACE FUNCTION prevent_self_admin()
RETURNS TRIGGER LANGUAGE plpgsql SECURITY DEFINER AS $$
BEGIN
  -- Only allow admin assignment via service role (not anon/authenticated)
  IF NEW.role = 'admin' AND OLD.role != 'admin' THEN
    IF current_setting('request.jwt.claims', true)::jsonb ->> 'role' != 'service_role' THEN
      RAISE EXCEPTION 'Cannot self-assign admin role';
    END IF;
  END IF;
  RETURN NEW;
END;
$$;

CREATE TRIGGER enforce_no_self_admin
  BEFORE UPDATE ON profiles
  FOR EACH ROW EXECUTE FUNCTION prevent_self_admin();

-- Auto-generate slug from title
CREATE OR REPLACE FUNCTION generate_startup_slug()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE
  base_slug TEXT;
  final_slug TEXT;
  counter   INTEGER := 0;
BEGIN
  base_slug := lower(regexp_replace(trim(NEW.title), '[^a-zA-Z0-9\s]', '', 'g'));
  base_slug := regexp_replace(base_slug, '\s+', '-', 'g');
  final_slug := base_slug;
  WHILE EXISTS (SELECT 1 FROM startup_ideas WHERE slug = final_slug AND id != COALESCE(NEW.id, uuid_generate_v4())) LOOP
    counter := counter + 1;
    final_slug := base_slug || '-' || counter;
  END LOOP;
  NEW.slug := final_slug;
  RETURN NEW;
END;
$$;

CREATE TRIGGER auto_slug_startup
  BEFORE INSERT ON startup_ideas
  FOR EACH ROW
  WHEN (NEW.slug IS NULL OR NEW.slug = '')
  EXECUTE FUNCTION generate_startup_slug();

-- 8. STARTUP RATINGS
-- ============================================================

CREATE TABLE startup_ratings (
  id          UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  startup_id  UUID NOT NULL REFERENCES startup_ideas(id) ON DELETE CASCADE,
  investor_id UUID NOT NULL REFERENCES profiles(id) ON DELETE CASCADE,
  rating      INTEGER NOT NULL CHECK (rating BETWEEN 1 AND 5),
  feedback    TEXT,
  tags        TEXT[],   -- ['Strong Problem', 'Interesting Market', ...]
  created_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at  TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  CONSTRAINT one_rating_per_investor UNIQUE (startup_id, investor_id)
);

-- Recalculate avg_rating on the startup after each insert/update/delete
CREATE OR REPLACE FUNCTION update_startup_avg_rating()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  UPDATE startup_ideas
  SET
    avg_rating   = COALESCE((SELECT AVG(rating) FROM startup_ratings WHERE startup_id = COALESCE(NEW.startup_id, OLD.startup_id)), 0),
    rating_count = (SELECT COUNT(*) FROM startup_ratings WHERE startup_id = COALESCE(NEW.startup_id, OLD.startup_id))
  WHERE id = COALESCE(NEW.startup_id, OLD.startup_id);
  RETURN NULL;
END;
$$;

CREATE TRIGGER recalc_avg_rating
  AFTER INSERT OR UPDATE OR DELETE ON startup_ratings
  FOR EACH ROW EXECUTE FUNCTION update_startup_avg_rating();

-- Update posts.comments_count
CREATE OR REPLACE FUNCTION update_comments_count()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  IF TG_OP = 'INSERT' AND NEW.content_type = 'post' THEN
    UPDATE posts SET comments_count = comments_count + 1 WHERE id = NEW.content_id;
  ELSIF TG_OP = 'DELETE' AND OLD.content_type = 'post' THEN
    UPDATE posts SET comments_count = GREATEST(comments_count - 1, 0) WHERE id = OLD.content_id;
  END IF;
  RETURN NULL;
END;
$$;

CREATE TRIGGER sync_comment_count
  AFTER INSERT OR DELETE ON comments
  FOR EACH ROW EXECUTE FUNCTION update_comments_count();

-- 12. FOLLOWS
-- ============================================================

CREATE TABLE follows (
  id           UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  follower_id  UUID NOT NULL REFERENCES profiles(id) ON DELETE CASCADE,
  following_id UUID NOT NULL REFERENCES profiles(id) ON DELETE CASCADE,
  created_at   TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  CONSTRAINT no_self_follow CHECK (follower_id != following_id),
  CONSTRAINT unique_follow UNIQUE (follower_id, following_id)
);

-- Sync follower/following counts
CREATE OR REPLACE FUNCTION update_follow_counts()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  IF TG_OP = 'INSERT' THEN
    UPDATE profiles SET following_count = following_count + 1 WHERE id = NEW.follower_id;
    UPDATE profiles SET followers_count = followers_count + 1 WHERE id = NEW.following_id;
    -- Update startup investor_interest_count when investor follows founder
    UPDATE startup_ideas
    SET investor_interest_count = investor_interest_count + 1
    WHERE founder_id = NEW.following_id
      AND EXISTS (SELECT 1 FROM investor_profiles ip JOIN profiles p ON p.id = ip.profile_id WHERE p.id = NEW.follower_id);
  ELSIF TG_OP = 'DELETE' THEN
    UPDATE profiles SET following_count = GREATEST(following_count - 1, 0) WHERE id = OLD.follower_id;
    UPDATE profiles SET followers_count = GREATEST(followers_count - 1, 0) WHERE id = OLD.following_id;
  END IF;
  RETURN NULL;
END;
$$;

CREATE TRIGGER sync_follow_counts
  AFTER INSERT OR DELETE ON follows
  FOR EACH ROW EXECUTE FUNCTION update_follow_counts();

CREATE TABLE conversations (
  id         UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE TABLE conversation_members (
  conversation_id UUID NOT NULL REFERENCES conversations(id) ON DELETE CASCADE,
  user_id         UUID NOT NULL REFERENCES profiles(id) ON DELETE CASCADE,
  joined_at       TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  last_read_at    TIMESTAMPTZ,
  PRIMARY KEY (conversation_id, user_id)
);

-- Helper: get or create a 1-to-1 conversation between two users
CREATE OR REPLACE FUNCTION get_or_create_conversation(user_a UUID, user_b UUID)
RETURNS UUID LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  conv_id UUID;
BEGIN
  -- Find existing direct conversation
  SELECT cm1.conversation_id INTO conv_id
  FROM conversation_members cm1
  JOIN conversation_members cm2 ON cm1.conversation_id = cm2.conversation_id
  WHERE cm1.user_id = user_a AND cm2.user_id = user_b
  LIMIT 1;

  IF conv_id IS NULL THEN
    INSERT INTO conversations DEFAULT VALUES RETURNING id INTO conv_id;
    INSERT INTO conversation_members (conversation_id, user_id) VALUES (conv_id, user_a), (conv_id, user_b);
  END IF;

  RETURN conv_id;
END;
$$;

-- Touch conversation.updated_at on new message
CREATE OR REPLACE FUNCTION touch_conversation()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  UPDATE conversations SET updated_at = NOW() WHERE id = NEW.conversation_id;
  RETURN NULL;
END;
$$;

CREATE TRIGGER on_new_message
  AFTER INSERT ON messages
  FOR EACH ROW EXECUTE FUNCTION touch_conversation();

-- Auto-generate slug for news
CREATE OR REPLACE FUNCTION generate_news_slug()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
DECLARE
  base_slug TEXT;
  final_slug TEXT;
  counter   INTEGER := 0;
BEGIN
  base_slug := lower(regexp_replace(trim(NEW.title), '[^a-zA-Z0-9\s]', '', 'g'));
  base_slug := regexp_replace(base_slug, '\s+', '-', 'g');
  final_slug := base_slug;
  WHILE EXISTS (SELECT 1 FROM news_articles WHERE slug = final_slug AND id != COALESCE(NEW.id, uuid_generate_v4())) LOOP
    counter := counter + 1;
    final_slug := base_slug || '-' || counter;
  END LOOP;
  NEW.slug := final_slug;
  IF NEW.status = 'published' AND NEW.published_at IS NULL THEN
    NEW.published_at := NOW();
  END IF;
  RETURN NEW;
END;
$$;

CREATE TRIGGER auto_slug_news
  BEFORE INSERT ON news_articles
  FOR EACH ROW
  WHEN (NEW.slug IS NULL OR NEW.slug = '')
  EXECUTE FUNCTION generate_news_slug();

-- Set published_at on status change to published
CREATE OR REPLACE FUNCTION set_published_at()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  IF NEW.status = 'published' AND OLD.status != 'published' AND NEW.published_at IS NULL THEN
    NEW.published_at := NOW();
  END IF;
  RETURN NEW;
END;
$$;

CREATE TRIGGER on_publish_news
  BEFORE UPDATE ON news_articles
  FOR EACH ROW EXECUTE FUNCTION set_published_at();

-- Public startup cards view (joins profile info)
CREATE OR REPLACE VIEW public_startups AS
SELECT
  s.id,
  s.title,
  s.slug,
  s.short_pitch,
  s.industry,
  s.stage,
  s.funding_required,
  s.cover_image_url,
  s.avg_rating,
  s.rating_count,
  s.investor_interest_count,
  s.visibility,
  s.created_at,
  p.full_name   AS founder_name,
  p.avatar_url  AS founder_avatar,
  p.username    AS founder_username
FROM startup_ideas s
JOIN profiles p ON p.id = s.founder_id
WHERE s.visibility = 'public';

-- Published news view
CREATE OR REPLACE VIEW published_news AS
SELECT
  n.id,
  n.title,
  n.slug,
  n.summary,
  n.cover_image_url,
  n.category,
  n.source_name,
  n.source_url,
  n.published_at,
  p.full_name AS author_name
FROM news_articles n
LEFT JOIN profiles p ON p.id = n.created_by
WHERE n.status = 'published'
ORDER BY n.published_at DESC;

-- Feed view: posts with author info
CREATE OR REPLACE VIEW feed_posts AS
SELECT
  po.id,
  po.post_type,
  po.content,
  po.image_url,
  po.tags,
  po.likes_count,
  po.comments_count,
  po.saves_count,
  po.created_at,
  po.startup_id,
  p.id          AS author_id,
  p.full_name   AS author_name,
  p.avatar_url  AS author_avatar,
  p.username    AS author_username,
  p.role        AS author_role,
  COALESCE(ip.verified, FALSE) AS author_verified
FROM posts po
JOIN profiles p ON p.id = po.author_id
LEFT JOIN investor_profiles ip ON ip.profile_id = p.id
ORDER BY po.created_at DESC;

-- Admin stats view
CREATE OR REPLACE VIEW admin_stats AS
SELECT
  (SELECT COUNT(*) FROM profiles WHERE role = 'student')        AS total_students,
  (SELECT COUNT(*) FROM profiles WHERE role = 'investor')       AS total_investors,
  (SELECT COUNT(*) FROM profiles WHERE role = 'business_owner') AS total_business_owners,
  (SELECT COUNT(*) FROM profiles WHERE role = 'general_user')   AS total_general_users,
  (SELECT COUNT(*) FROM profiles)                               AS total_users,
  (SELECT COUNT(*) FROM startup_ideas WHERE visibility = 'public') AS public_startups,
  (SELECT COUNT(*) FROM news_articles WHERE status = 'published')  AS published_news,
  (SELECT COUNT(*) FROM news_articles WHERE status = 'draft')      AS draft_news,
  (SELECT COUNT(*) FROM posts)                                      AS total_posts,
  (SELECT COUNT(*) FROM messages)                                   AS total_messages;

-- Helper: get own profile id
CREATE OR REPLACE FUNCTION own_profile_id()
RETURNS UUID LANGUAGE sql STABLE SECURITY DEFINER AS $$
  SELECT id FROM profiles WHERE user_id = auth.uid() LIMIT 1;
$$;

-- Helper: own role
CREATE OR REPLACE FUNCTION own_role()
RETURNS TEXT LANGUAGE sql STABLE SECURITY DEFINER AS $$
  SELECT role::TEXT FROM profiles WHERE user_id = auth.uid() LIMIT 1;
$$;
